- 확장자명은 .sm
- 구조
    - 1. 표지 : 정보의 환경 정보
    - 2. 차례 : 정보의 위치 정보
    - n. 내용 : 네트워크 설정, 뉴런 설정, 시냅스 설정 + 가중치 등등 여러 데이터를 저장하고 있는다.
    - n + 1. 기타 : 부가적인 정보

for (auto &n : neuron_contexts_)
                {
                    n->neuron->step(n->current, n->spike);
                    std::fill(n->current.begin(), n->current.end(), 0.0f);
                }
    
                // 4. 출력 스파이크 추출
                outputs.resize(output_neuron_ids_.size());
                for (std::size_t idx = 0; idx < output_neuron_ids_.size(); ++idx)
                {
                    outputs[idx].copy_from(neuron_contexts_[output_neuron_ids_[idx]]->spike);
                }
            }
    
            // 단일 입력/출력 하위 호환 오버로딩 (기존 코드 지원)
            void step(const F32Buffer &input, BitBuffer &output)
            {
                std::vector<F32Buffer> inputs;
                if (!input.empty())
                {
                    inputs.push_back(input);
                }
    
                std::vector<BitBuffer> outputs;
                step(inputs, outputs);
    
                if (!outputs.empty())
                {
                    output.copy_from(outputs[0]);
                }
            }
    
            [[nodiscard]] NeuronContext &get_neuron_context(std::size_t id)
            {
                return *neuron_contexts_[id];
            }
    
            [[nodiscard]] SynapseContext &get_synapse_context(std::size_t id)
                return *synapse_contexts_[id];
            }

            [[nodiscard]] const std::vector<std::size_t> &input_ids() const noexcept { return input_neuron_ids_; }
            [[nodiscard]] const std::vector<std::size_t> &output_ids() const noexcept { return output_neuron_ids_; }
        };
